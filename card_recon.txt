import { useCallback, useEffect, useState } from 'react'
import { Navigate, Routes, Route, useNavigate } from 'react-router-dom'
import Landing from './pages/Landing.jsx'
import Specify from './pages/Specify.jsx'
import Checkout from './pages/Checkout.jsx'
import Track from './pages/Track.jsx'
import Paid from './pages/Paid.jsx'
import ShortLink from './pages/ShortLink.jsx'
import Account from './pages/Account.jsx'
import Categories from './pages/Categories.jsx'
import Orders from './pages/Orders.jsx'
import Settings from './pages/Settings.jsx'
import Terms from './pages/Terms.jsx'
import Notifications from './pages/Notifications.jsx'
import PushBridge from './lib/PushBridge.jsx'
import LaunchSplash from './components/LaunchSplash.jsx'
import TabBar from './components/TabBar.jsx'
import { AccountProvider } from './lib/AccountContext.jsx'
import { takePaying } from './lib/paying'
import { useQuery } from '@tanstack/react-query'
import { getCatalogue } from './api/client'
import { rememberServerImages } from './lib/images'
import Directions from './pages/Directions.jsx'
import Board from './pages/Board.jsx'

export default function App() {
  // The opening animation, on top of the app while it loads. Only in the
  // phone app, only on a cold start; on the website it is gone at once.
  const [opening, setOpening] = useState(true)
  const opened = useCallback(() => setOpening(false), [])
  const navigate = useNavigate()

  // The catalogue, once at the start, so every screen's photos come from
  // the server: the same query the shop pages use, so it is shared.
  const { data: catalogue } = useQuery({
    queryKey: ['catalogue'],
    queryFn: () => getCatalogue().then(r => r.data),
  })
  const [, setPhotosReady] = useState(0)
  useEffect(() => {
    if (!catalogue) return
    rememberServerImages(catalogue)
    setPhotosReady(n => n + 1)
  }, [catalogue])

  // Back from Paystack. Either the website's return page hands over
  // through printoctos://paid, or the customer closes the panel
  // themselves; both land on the confirmation, which asks the server
  // what really happened.
  useEffect(() => {
    if (!window.Capacitor?.isNativePlatform?.()) return undefined
    const subs = []
    ;(async () => {
      const { App: Native } = await import('@capacitor/app')
      const { Browser } = await import('@capacitor/browser')
      subs.push(await Native.addListener('appUrlOpen', ({ url }) => {
        if (!url.startsWith('printoctos://paid')) return
        takePaying()
        const reference = new URL(url.replace('printoctos://', 'https://app/')).searchParams.get('reference') || ''
        Browser.close().catch(() => {})
        navigate(`/paid?reference=${encodeURIComponent(reference)}`)
      }))
      subs.push(await Browser.addListener('browserFinished', () => {
        const reference = takePaying()
        if (reference) navigate(`/paid?reference=${encodeURIComponent(reference)}&closed=1`)
      }))
    })()
    return () => subs.forEach(s => s.remove())
  }, [navigate])

  return (
    <AccountProvider>
      <Routes>
        <Route path="/" element={<Landing />} />
        <Route path="/categories" element={<Categories />} />
        <Route path="/orders" element={<Orders />} />
        {/* More became the Account tab; old links still arrive. */}
        <Route path="/more" element={<Navigate to="/account" replace />} />
        <Route path="/settings" element={<Settings />} />
        <Route path="/terms" element={<Terms />} />
        <Route path="/account" element={<Account />} />
        <Route path="/notifications" element={<Notifications />} />
        <Route path="/service/:serviceId" element={<Specify />} />
        <Route path="/checkout" element={<Checkout />} />
        <Route path="/track" element={<Track />} />
        <Route path="/board/:branch" element={<Board />} />
        <Route path="/track/:orderNumber" element={<Track />} />
        {/* Where Paystack sends them afterwards, whether they paid or
            changed their mind, which is why the page asks before it
            celebrates. */}
        <Route path="/paid" element={<Paid />} />
        {/* The link in the text message, traded for the real token. */}
        <Route path="/t/:code" element={<ShortLink />} />
        <Route path="/directions" element={<Directions />} />
      </Routes>
      <TabBar />
      <PushBridge />
      {opening && <LaunchSplash onDone={opened} />}
    </AccountProvider>
  )
}

import axios from 'axios'

// No auth header, ever. The storefront is unauthenticated by design:
// what stands in for a login is the token on each order, passed with
// the requests that need it.
const client = axios.create({
  baseURL: import.meta.env.VITE_API_URL || 'http://localhost:8000',
  headers: { 'Content-Type': 'application/json' },
})

export const getCatalogue = () =>
  client.get('/api/v1/storefront/catalogue/')

export const createOrder = () =>
  client.post('/api/v1/storefront/orders/', {})

export const getOrder = (orderNumber, token) =>
  client.get(`/api/v1/storefront/orders/${orderNumber}/`, { params: { token } })

export const updateOrder = (orderNumber, token, changes) =>
  client.patch(`/api/v1/storefront/orders/${orderNumber}/`, { token, ...changes })

export const identify = (orderNumber, token, details) =>
  client.post(`/api/v1/storefront/orders/${orderNumber}/identify/`, { token, ...details })

export const setCode = (orderNumber, token) =>
  client.post(`/api/v1/storefront/orders/${orderNumber}/code/`, { token })

export const startPayment = (orderNumber, token, callbackUrl) =>
  client.post(`/api/v1/storefront/orders/${orderNumber}/pay/`, {
    token,
    callback_url: callbackUrl,
  })

export const uploadFile = (orderNumber, token, file, lineId, onProgress) => {
  const form = new FormData()
  form.append('token', token)
  form.append('file', file)
  // Which item this artwork is for. An order can hold a banner, flyers
  // and programmes at once, and the banner's file is not the flyer's.
  if (lineId) form.append('line_id', lineId)
  // A bare axios call rather than the shared client. That one sets a
  // JSON content type on every request, and a multipart body sent as
  // JSON arrives at the server with no fields in it â€” no token, so the
  // order is not found, and the error says nothing about the cause.
  return axios.post(
    `${client.defaults.baseURL}/api/v1/storefront/orders/${orderNumber}/file/`,
    form,
    {
      // Real progress, not a guess. On a slow connection most of the
      // wait is the bytes leaving the phone, and that is worth showing
      // rather than pretending.
      onUploadProgress: event => {
        if (!onProgress || !event.total) return
        onProgress(Math.round((event.loaded / event.total) * 100))
      },
    },
  )
}

export const getFile = (orderNumber, token, lineId) =>
  client.get(`/api/v1/storefront/orders/${orderNumber}/file/`, {
    params: { token, ...(lineId ? { line_id: lineId } : {}) },
  })

export const acceptWarning = (orderNumber, token, lineId) =>
  client.patch(`/api/v1/storefront/orders/${orderNumber}/file/`, {
    token,
    line_id: lineId,
    warning_accepted: true,
  })

export const getBranchOptions = (orderNumber, token, location) =>
  client.get(`/api/v1/storefront/orders/${orderNumber}/branches/`, {
    params: {
      token,
      ...(location ? {
        latitude: location.latitude,
        longitude: location.longitude,
      } : {}),
    },
  })

/**
 * Judges artwork against a size, and commits nothing.
 *
 * Comes back with a signed verdict that the commit will check against
 * the file and the size it was judged at â€” so a customer cannot pass a
 * good file and then send something else.
 */
export const checkArtwork = (file, serviceId, specifications, onProgress) => {
  const form = new FormData()
  form.append('file', file)
  form.append('service', serviceId)
  form.append('specifications', JSON.stringify(specifications))

  return axios.post(
    `${client.defaults.baseURL}/api/v1/storefront/check/`,
    form,
    {
      onUploadProgress: event => {
        if (!onProgress || !event.total) return
        onProgress(Math.round((event.loaded / event.total) * 100))
      },
    },
  )
}

/**
 * What a line would cost, without committing it. Finishing chosen
 * (lamination, binding, an envelope) is priced with it.
 */
export const quoteLine = (serviceId, quantity, specifications, addons = []) =>
  client.post('/api/v1/storefront/quote/', {
    service: serviceId,
    quantity,
    specifications,
    addons,
  })

/**
 * Continue. The moment anything becomes real.
 *
 * The key makes a retry safe: a commit whose response is lost is a
 * normal event on mobile data, and pressing again must not buy the
 * banner twice.
 */
export const commitLine = ({
  key, orderNumber, token, serviceId, quantity, specifications, addons = [], artworkToken,
}) =>
  client.post('/api/v1/storefront/commit/', {
    idempotency_key: key,
    ...(orderNumber ? { order: orderNumber, token } : {}),
    service: serviceId,
    quantity,
    specifications,
    addons,
    ...(artworkToken ? { artwork_token: artworkToken } : {}),
  })

/**
 * The way back in without the link. The order number and the phone
 * they gave â€” neither opens anything alone.
 */
export const findOrder = (orderNumber, phone) =>
  client.post('/api/v1/storefront/find/', { order: orderNumber, phone })

/**
 * The short code from a text message, traded for the real token.

export const openShortLink = code =>
  client.get(`/api/v1/storefront/t/${code}/`)

/**
 * What to show somebody who has just paid, from the Paystack
 * reference alone. Needs nothing the browser has kept.
 */
export const confirmPayment = reference =>
  client.get(`/api/v1/storefront/confirm/${reference}/`)

/**
 * The app's session, sent with every request once signed in. The website
 * has no accounts, so there it is never set.
 */
export const setAppToken = token => {
  if (token) client.defaults.headers.common.Authorization = `App ${token}`
  else delete client.defaults.headers.common.Authorization
}

  // Several files, each with its sets, added to the order in one go.
export const commitBatch = ({ key, orderNumber, token, specifications, files }) =>
