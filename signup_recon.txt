"""
Somebody who has ordered online but is not yet a customer.

A first visit asks for a phone number and a first name, nothing more.
Asking a stranger to register before they know what a job costs loses
them, and most first visits never come back â€” writing every one of them
into CustomerProfile would fill the customer table with people the shop
has met once.

So a first order creates a Lead. When the same number orders again, the
lead has earned a profile, and the second visit is where the surname and
a PIN are asked for. Offered, not forced: a wall at the moment they came
back is the wrong moment.
"""

from django.db import models

from apps.core.models import AuditModel


class Lead(AuditModel):
    phone = models.CharField(max_length=20, unique=True, db_index=True)
    first_name = models.CharField(max_length=100)

    # Set when the lead becomes a real customer. The orders stay attached
    # to the lead; the profile is what the shop deals with afterwards.
    customer = models.OneToOneField(
        'customers.CustomerProfile',
        on_delete=models.SET_NULL,
        null=True, blank=True,
        related_name='lead',
        help_text='Set on the visit that turns this lead into a customer.',
    )

    # Protects history, never the job itself. Someone who has just
    # ordered reaches that order by its link with no code at all â€” the
    # code is what stops a stranger reading everything this number has
    # ever ordered, by knowing the number.
    #
    # Hashed, and never readable again. Losing it means being sent a new
    # one, which is the same experience for the customer and leaves
    # nothing in the database worth stealing.
    code = models.CharField(
        max_length=128, blank=True,
        help_text='Hashed. Set when the customer chooses to keep their history.',
    )

    # The 5% is a reason to set a code, paid once to each person. Held
    # here rather than against the code itself, so clearing a code and
    # setting another earns nothing new.
    code_discount_used = models.BooleanField(default=False)

    order_count = models.PositiveIntegerField(default=0)
    last_ordered_at = models.DateTimeField(null=True, blank=True)

    class Meta:
        ordering = ['-created_at']

    def __str__(self):
        return f"{self.first_name} â€” {self.phone}"

    @property
    def is_returning(self):
        """A second order is what earns a profile."""
        return self.order_count > 0

    @property
    def has_code(self):
        return bool(self.code)

    def set_code(self, raw_code):
        """Hashed on the way in. Never stored as typed."""
        from django.contrib.auth.hashers import make_password
        self.code = make_password(raw_code)

    def check_code(self, raw_code):
        from django.contrib.auth.hashers import check_password
        return bool(self.code) and check_password(raw_code, self.code)

"""
Sign up and sign in for the Print Octos app.

POST account/code/      { phone }                          text a code
POST account/verify/    { phone, code, first_name?, device? }  sign in or up
GET  account/me/        (App token)                        who is signed in
POST account/signout/   (App token)                        end this session
"""

from rest_framework import status
from rest_framework.permissions import AllowAny
from rest_framework.response import Response
from rest_framework.throttling import ScopedRateThrottle
from rest_framework.views import APIView

from apps.storefront.auth import AppSessionAuthentication
from apps.storefront.services.accounts import (
    AccountRefused, mask, send_code, sign_out, verify_code,
)


def _account(lead):
    from apps.storefront.account_profile import profile_for
    profile = profile_for(lead)
    return {
        'first_name': profile.get('first_name') or lead.first_name,
        'phone': mask(lead.phone),
        # The customer's own number, for their own orders: shown masked,
        # sent in full to the signed-in app that owns it.
        'number': lead.phone,
        'order_count': lead.order_count,
        'member_since': lead.created_at.isoformat(),
        # Who they are to the shop, and whether the short registration
        # is still needed.
        'profile': profile,
    }


class _Public(APIView):
    permission_classes = [AllowAny]
    authentication_classes = []
    throttle_classes = [ScopedRateThrottle]
    # The same limit as guessing a customer code: a few a minute per device.
    throttle_scope = 'storefront_code'


class AccountCodeView(_Public):
    def post(self, request):
        try:
            phone = send_code(request.data.get('phone'))
        except AccountRefused as refused:
            return Response({'detail': str(refused)}, status=status.HTTP_400_BAD_REQUEST)
        from apps.storefront.services.accounts import find_lead
        return Response({
            'sent_to': mask(phone),
            # So the app knows whether to ask for a first name.
            'new_customer': find_lead(phone) is None,
        })


class AccountVerifyView(_Public):
    def post(self, request):
        try:
            lead, token = verify_code(
                request.data.get('phone'), request.data.get('code'),
                first_name=request.data.get('first_name', ''),
                device=request.data.get('device', ''),
            )
        except AccountRefused as refused:
            return Response({'detail': str(refused)}, status=status.HTTP_400_BAD_REQUEST)
        return Response({'token': token, 'account': _account(lead)}, status=status.HTTP_201_CREATED)


class _SignedIn(APIView):
    authentication_classes = [AppSessionAuthentication]
    permission_classes = [AllowAny]
    throttle_classes = [ScopedRateThrottle]
    throttle_scope = 'storefront'

    def session(self, request):
        from apps.storefront.models import AppSession
        return request.auth if isinstance(request.auth, AppSession) else None

    def refused(self):
        return Response({'detail': 'Sign in to continue.'}, status=status.HTTP_401_UNAUTHORIZED)


class AccountMeView(_SignedIn):
    def get(self, request):
        session = self.session(request)
        if session is None:
            return self.refused()
        return Response(_account(session.lead))

    def patch(self, request):
        """Their first name, as they want it written. The number is the account and does not change here."""
        session = self.session(request)
        if session is None:
            return self.refused()
        name = (request.data.get('first_name') or '').strip()
        if not name:
            return Response({'detail': 'Tell us your first name.'}, status=status.HTTP_400_BAD_REQUEST)
        session.lead.first_name = name[:100]
        session.lead.save(update_fields=['first_name'])
        if session.lead.customer_id:
            session.lead.customer.first_name = name[:100]
            session.lead.customer.save(update_fields=['first_name'])
        return Response(_account(session.lead))


class AccountSignOutView(_SignedIn):
    def post(self, request):
        session = self.session(request)
        if session is None:
            return self.refused()
        sign_out(session)
        return Response({'signed_out': True})

def leads_for(lead):
    """
    Every Lead on this number, however the website wrote it down: an
    order placed as "024 499 9888" and one as "0244999888" are the same
    customer's.
    """
    from apps.storefront.models import Lead

    phone = normalise_phone(lead.phone) or lead.phone
    same = [
        other for other in Lead.objects.filter(phone__contains=phone[-4:])
        if normalise_phone(other.phone) == phone
    ]
    return same or [lead]


# The customer's words for where an order has got to.
_STAGE = {
    'RECEIVED': 'inspecting',
    'IN_PRODUCTION': 'press',
    'FINISHING': 'finishing',
    'QUALITY_CHECK': 'finishing',
    'DONE': 'ready',
}


def _stage(order):
    job = order.job
    if job is None:
        return 'paid'
    if job.handover_state == 'HANDED_OVER':
        return 'collected'
    return _STAGE.get(job.work_state, 'paid')


def orders_for(lead):
    """
    The customer's paid orders, newest first: active ones with their stage
    and time left, collected ones with when. Cancelled work is left out;
    carts never paid for are not orders.
    """
    from apps.jobs.models import JobStatusLog
    from apps.jobs.services.estimates import estimate
    from apps.storefront.models import OnlineOrder

    orders = (
        OnlineOrder.objects
        .filter(lead__in=leads_for(lead), paid_at__isnull=False)
        .select_related('job', 'branch')
        .order_by('-paid_at')
    )

    active, collected = [], []
    for order in orders:
        job = order.job
        if job is not None and job.status in ('CANCELLED', 'VOIDED'):
            continue

        lines = order.line_items or []
        first = lines[0].get('service_name', 'Order') if lines else 'Order'
        item = {
            'order_number': order.order_number,
            # Their own order: the link that opens its tracking page.
            'token': order.access_token,
            'summary': first if len(lines) <= 1 else f'{first} and {len(lines) - 1} more',
            'total': f'{order.total:.2f}',
            'paid_at': order.paid_at.isoformat(),
            'branch': order.branch.name if order.branch_id else None,
            'promised_for': order.promised_for.isoformat() if order.promised_for else None,
            'stage': _stage(order),
        }

        if item['stage'] == 'collected':
            at = (
                JobStatusLog.objects
                .filter(job=job, to_status='HANDED_OVER')
                .order_by('transitioned_at')
                .values_list('transitioned_at', flat=True)
                .first()
            )
            item['collected_at'] = (at or job.updated_at).isoformat()
            collected.append(item)
        else:
            try:
                found = estimate(job) if job is not None else None
            except Exception:
                found = None
            item['estimate'] = found.as_dict() if found else None
            active.append(item)

    return {'active': active, 'collected': collected}

class AccountDeviceView(_SignedIn):
    """POST account/device/ { token, platform }: where to push for this phone."""

    def post(self, request):
        from django.utils import timezone
        from apps.storefront.models import DeviceToken
        session = self.session(request)
        if session is None:
            return self.refused()
        token = (request.data.get('token') or '').strip()
        if not token:
            return Response({'detail': 'No device token.'}, status=status.HTTP_400_BAD_REQUEST)
        DeviceToken.objects.update_or_create(
            token=token[:255],
            defaults={
                'session': session,
                'platform': (request.data.get('platform') or '')[:10],
                'last_seen_at': timezone.now(),
            },
        )
        return Response({'registered': True})


class AccountNotificationsView(_SignedIn):
    """
    GET account/notifications/: the inbox, newest first, with the unread
    count. Each message about an order carries that order's product (name,
    detail, add-ons), what was paid and how, so the app can show it as one
    card per order without asking again.
    """

    @staticmethod
    def _order(order):
        from apps.storefront.services.accounts import _detail
        lines = order.line_items or []
        first = lines[0] if lines else {}
        return {
            'order_number': order.order_number,
            'token': order.access_token,
            'product': {
                'name': first.get('service_name', ''),
                'base_name': first.get('base_name') or first.get('service_name', ''),
                'detail': _detail(first) if first else '',
                'addons': [a.get('service_name', '') for a in (first.get('addons') or [])],
                'more': max(0, len(lines) - 1),
            } if first else None,
            'total': f'{order.total:.2f}',
            'payment_channel': getattr(order, 'payment_channel', '') or '',
        }

    def get(self, request):
        from apps.storefront.models import AppNotification
        from apps.storefront.services.accounts import leads_for
        session = self.session(request)
        if session is None:
            return self.refused()
        notes = (
            AppNotification.objects
            .filter(lead__in=leads_for(session.lead))
            .select_related('order')[:50]
        )
        items = []
        for n in notes:
            item = {
                'id': n.id,
                'kind': n.kind,
                'title': n.title,
                'body': n.body,
                'at': n.created_at.isoformat(),
                'read': n.read_at is not None,
                'order_number': None,
                'token': None,
            }
            if n.order_id:
                item.update(self._order(n.order))
            items.append(item)
        return Response({
            'unread': sum(1 for n in notes if n.read_at is None),
            'items': items,
        })


class AccountNotificationsReadView(_SignedIn):
    """POST account/notifications/read/: everything in the inbox marked read."""

    def post(self, request):
        from django.utils import timezone
        from apps.storefront.models import AppNotification
        from apps.storefront.services.accounts import leads_for
        session = self.session(request)
        if session is None:
            return self.refused()
        AppNotification.objects.filter(
            lead__in=leads_for(session.lead), read_at__isnull=True,
        ).update(read_at=timezone.now())
        return Response({'read': True})
