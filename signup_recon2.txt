"""
Who an app account is: the customer behind the number.

Signing in proves a number; this says who holds it. A number the counter
already knows is the counter's customer, so Eric Nortey, a regular since
before the app, is welcomed back rather than signed up again. A number
nobody knows yet is asked, once, for a short registration, which makes
a proper customer the counter recognises from the first visit.

POST account/register/ (App token)
    { kind: PERSONAL | BUSINESS, title, gender, first_name, last_name,
      business_name?, address? }

Completes the customer behind the number, or makes one: never a second.
"""

from django.db import transaction
from django.db.models import Q
from rest_framework import status
from rest_framework.response import Response

TITLES = {'MR', 'MRS', 'MS', 'DR', 'PROF', 'OTHER'}
GENDERS = {'MALE', 'FEMALE'}
ORG_TYPES = ('BUSINESS', 'INSTITUTION')


def customer_for(lead):
    """The customer behind this lead: linked already, or found by any of their numbers (and linked)."""
    from apps.core.phones import lookup_forms
    from apps.customers.models import CustomerProfile

    if lead.customer_id:
        return lead.customer
    forms = lookup_forms(lead.phone)
    found = (CustomerProfile.objects
             .filter(Q(phone__in=forms) | Q(secondary_phone__in=forms))
             .order_by('created_at').first())
    if found is not None:
        lead.customer = found
        lead.save(update_fields=['customer'])
    return found


def profile_for(lead):
    """What the app shows and whether the short registration is still needed."""
    c = customer_for(lead)
    if c is None:
        return {'complete': False, 'kind': None}
    business = c.customer_type in ORG_TYPES
    complete = bool(c.first_name and c.last_name and c.title and (
        (c.company_name and c.address) if business else c.gender))
    return {
        'complete': complete,
        'kind': 'BUSINESS' if business else 'PERSONAL',
        'title': c.title or '',
        'first_name': c.first_name or '',
        'last_name': c.last_name or '',
        'gender': c.gender or '',
        'business_name': c.company_name or '' if business else '',
        'address': c.address or '' if business else '',
    }


def register(lead, data):
    """Complete the customer behind the number, or make one. Returns problems, or None."""
    from apps.customers.models import CustomerProfile
    from apps.customers.utils import normalise_phone

    kind = (data.get('kind') or 'PERSONAL').upper()
    title = (data.get('title') or '').upper()
    gender = (data.get('gender') or '').upper()
    first = (data.get('first_name') or '').strip()[:100]
    last = (data.get('last_name') or '').strip()[:100]
    business = (data.get('business_name') or '').strip()[:200]
    address = (data.get('address') or '').strip()[:300]

    if kind not in ('PERSONAL', 'BUSINESS'):
        return 'Choose personal or business.'
    if title not in TITLES:
        return 'Choose a title.'
    if not first or not last:
        return 'Tell us your first and last name.'
    if kind == 'PERSONAL' and gender not in GENDERS:
        return 'Choose male or female.'
    if kind == 'BUSINESS' and (not business or not address):
        return 'Tell us the businessâ€™s name and where it is.'

    with transaction.atomic():
        c = customer_for(lead)
        if c is None:
            c = CustomerProfile(phone=normalise_phone(lead.phone),
                                customer_type='BUSINESS' if kind == 'BUSINESS' else 'INDIVIDUAL')
        elif kind == 'BUSINESS' and c.customer_type not in ORG_TYPES:
            # A known person registering a business on their own number:
            # the counter links people to businesses; the app keeps them
            # as the person they already are.
            return ('This number is already registered to you personally. '
                    'To order for a business, choose it at the counter, or use the businessâ€™s own number.')
        c.title = title
        c.first_name = first
        c.last_name = last
        if kind == 'PERSONAL':
            c.gender = gender
        else:
            c.company_name = business
            c.address = address
        c.save()
        lead.customer = c
        lead.first_name = first
        lead.save(update_fields=['customer', 'first_name'])
    return None


def register_view(view, request):
    session = view.session(request)
    if session is None:
        return view.refused()
    problem = register(session.lead, request.data)
    if problem:
        return Response({'detail': problem}, status=status.HTTP_400_BAD_REQUEST)
    from apps.storefront.account_views import _account
    return Response(_account(session.lead))

function SignIn() {
  const navigate = useNavigate()
  const [params] = useSearchParams()
  const { signedIn } = useAccount()
  const toast = useToast()
  const [step, setStep] = useState('phone')
  const [phone, setPhone] = useState('')
  const [sentTo, setSentTo] = useState('')
  const [isNew, setIsNew] = useState(false)
  const [code, setCode] = useState('')
  const [firstName, setFirstName] = useState('')
  const [busy, setBusy] = useState(false)
  const [problem, setProblem] = useState('')

  async function send() {
    setBusy(true); setProblem('')
    try {
      const { data } = await accountCode(phone.trim())
      setSentTo(data.sent_to)
      setIsNew(data.new_customer)
      setStep('code')
    } catch (err) {
      setProblem(err.response?.data?.detail || 'We couldnâ€™t send a code just now. Try again in a moment.')
    } finally { setBusy(false) }
  }

  async function verify() {
    setBusy(true); setProblem('')
    try {
      const { data } = await accountVerify({
        phone: phone.trim(), code: code.trim(), first_name: firstName.trim(),
        device: navigator.userAgent.slice(0, 120),
      })
      await signedIn(data.token, data.account)
      // Someone the shop already knows goes straight in; anyone else
      // stays here, where the short registration now shows.
      if (data.account.profile?.complete !== false) {
        toast.show(`Welcome back, ${data.account.first_name}`, 'Good to see you again.')
        navigate(params.get('next') || '/', { replace: true })
      }
    } catch (err) {
      setProblem(err.response?.data?.detail || 'That didnâ€™t work. Check the code and try again.')
      setBusy(false)
    }
  }

  const input = `w-full px-3 py-3 rounded-lg border border-rule bg-white text-[1rem]
                 focus:outline-none focus:border-ink`

  return (
    <div>
      <h1 className="text-xl font-bold">Sign in or create your account</h1>
      <p className="mt-1.5 text-sm leading-relaxed text-body">
        Your number is your account. Weâ€™ll text you a code to check itâ€™s
        yours. New here? The same code sets you up.
      </p>

      {step === 'phone' && (
        <>
          <label className="block mt-5">
            <span className="block mb-1.5 text-xs text-body">Phone number</span>
            <input type="tel" inputMode="tel" autoComplete="tel" value={phone}
                   onChange={e => setPhone(e.target.value)} placeholder="024 000 0000"
                   onKeyDown={e => { if (e.key === 'Enter' && phone.trim()) send() }}
                   className={input} />
          </label>
          <button onClick={send} disabled={busy || !phone.trim()}
                  className="mt-4 w-full px-4 py-3 rounded-lg bg-farhat text-white text-sm
                             font-semibold disabled:opacity-40">
            {busy ? 'Sendingâ€¦' : 'Send me a code'}
          </button>
        </>
      )}

      {step === 'code' && (
        <>
          <p className="mt-5 text-sm">We texted a code to <strong>{sentTo}</strong>.</p>
          <label className="block mt-3">
            <span className="block mb-1.5 text-xs text-body">The 6-digit code</span>
            <input inputMode="numeric" autoComplete="one-time-code" maxLength={6}
                   value={code} onChange={e => setCode(e.target.value.replace(/\D/g, ''))}
                   className={`${input} tracking-[0.5em] text-center text-lg font-semibold`} />
          </label>
          {isNew && (
            <label className="block mt-3">
              <span className="block mb-1.5 text-xs text-body">Your first name</span>
              <input autoComplete="given-name" value={firstName}
                     onChange={e => setFirstName(e.target.value)} className={input} />
            </label>
          )}
          <button onClick={verify}
                  disabled={busy || code.length !== 6 || (isNew && !firstName.trim())}
                  className="mt-4 w-full px-4 py-3 rounded-lg bg-farhat text-white text-sm
                             font-semibold disabled:opacity-40">
            {busy ? 'Checkingâ€¦' : isNew ? 'Create my account' : 'Sign in'}
          </button>
          <button onClick={() => { setStep('phone'); setCode(''); setProblem('') }}
                  className="mt-3 w-full text-sm font-semibold text-body">
            Use a different number, or send the code again
          </button>
        </>
      )}

      {problem && <p className="mt-4 text-sm text-farhat">{problem}</p>}
    </div>
  )
}

  return new Date(iso).toLocaleDateString('en-GB', { day: 'numeric', month: 'short' })

function timeLeft(e) {
  if (!e) return null
  if (e.held_up) return 'a short delay, weâ€™re on it'
  if (e.running_over) return 'taking a little longer than usual'
  const m = Number(e.minutes_left) || 0
  if (m < 60) return `about ${m} min left`
  const h = Math.floor(m / 60)
  return `about ${h} hr${m % 60 ? ` ${m % 60} min` : ''} left`
}

      {isLoading && <p className="mt-4 text-sm text-body">Loading your ordersâ€¦</p>}
      {isError && <p className="mt-4 text-sm text-farhat">Your orders didnâ€™t load. Try again in a moment.</p>}

      {!isLoading && !isError && list.length === 0 && (
        <div className="mt-4 rounded-xl border border-dashed border-rule px-4 py-6 text-center">
          <p className="text-sm text-body">
            {tab === 'active' ? 'Nothing being made right now.' : 'Nothing collected yet.'}
          </p>
          <button onClick={() => navigate('/')}
                  className="mt-3 px-4 py-2 rounded-lg bg-farhat text-white text-sm font-semibold">
            Order something
          </button>
        </div>
      )}

      <div className="mt-3 space-y-2">
        {list.map(o => {
          const stage = STAGE[o.stage]
          const left = timeLeft(o.estimate)
          return (
            <button key={o.order_number} onClick={() => open(o)}
                    className="w-full text-left rounded-xl border border-rule px-4 py-3
                               hover:border-ink transition-colors">
              <div className="flex items-start justify-between gap-3">
                <p className="text-sm font-semibold leading-snug">{o.summary}</p>
                {tab === 'active' && stage ? (
                  <span className={`shrink-0 px-2 py-0.5 rounded-full text-[0.7rem] font-semibold ${stage.tone}`}>
                    {stage.label}
                  </span>
                ) : (
                  <span className="shrink-0 text-sm font-semibold tabular-nums">GHS {o.total}</span>
                )}
              </div>
              <p className="mt-1 text-xs text-body">
                {o.order_number}
                {tab === 'active'
                  ? (left ? ` Â· ${left}` : o.branch ? ` Â· ${o.branch}` : '')
                  : ` Â· collected ${when(o.collected_at)}`}
              </p>
            </button>
          )
        })}
      </div>
    </section>
